bridge.py 
# 1. Importation des bibliothèques nécessaires
import oracledb
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression

# 2. Informations de connexion à la base Oracle VM (Docker / Local)
conn_params = {
    "user": "system",
    "password": "votre_mot_de_passe",
    "dsn": "localhost:1521/xe"  # Mettez votre SID/Service Name
}

try:
    # Connexion à Oracle Database
    connection = oracledb.connect(**conn_params)
    cursor = connection.cursor()
    
    # Extraction des métriques insérées par oracle_setup.sql
    query = """
        SELECT id_mesure, used_space_mb, max_space_mb 
        FROM ts_space_history 
        WHERE tablespace_name = 'TS_ADMIN_IA'
        ORDER BY id_mesure ASC
    """
    cursor.execute(query)
    rows = cursor.fetchall()
    
    # Conversion en DataFrame Pandas
    df = pd.DataFrame(rows, columns=['Jour', 'Espace_Utilise', 'Max_Espace'])
    
    # 3. Entraînement du modèle de Machine Learning (Régression Linéaire)
    X = df[['Jour']].values
    y = df['Espace_Utilise'].values
    
    model = LinearRegression()
    model.fit(X, y)
    
    m = model.coef_[0]
    b = model.intercept_
    
    # 4. Calcul du Time-To-Failure (TTF) pour la limite de 50 MB
    limite = df['Max_Espace'].iloc[-1]
    jour_saturation = (limite - b) / m
    
    print(f"Vitesse d'écriture : {m:.2f} MB / jour")
    print(f"Alerte : Saturation prévue au Jour {jour_saturation:.2f}")
    
except Exception as e:
    print("Erreur de liaison :", e)
finally:
    if 'connection' in locals():
        connection.close()

Simulateur de Requête DBA Oracle

Les requêtes SQL exécutées directement sur le dictionnaire de données Oracle pour alimenter l'interface.

-- Requête d'extraction en temps réel 
SELECT 
    b.tablespace_name,
    ROUND(SUM(b.bytes) / 1024 / 1024, 2) as size_mb,
    ROUND(SUM(b.bytes - NVL(f.bytes, 0)) / 1024 / 1024, 2) as used_mb,
    ROUND((SUM(b.bytes - NVL(f.bytes, 0)) / SUM(b.bytes)) * 100, 2) as pct_used
FROM dba_data_files b
LEFT JOIN (
    SELECT tablespace_name, file_id, SUM(bytes) as bytes 
    FROM dba_free_space 
    GROUP BY tablespace_name, file_id
) f ON b.tablespace_name = f.tablespace_name AND b.file_id = f.file_id
GROUP BY b.tablespace_name;
